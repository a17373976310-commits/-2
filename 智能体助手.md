【判断标准】一个系统是否是智能体，需满足：
□ 能否感知环境（接收输入/读取状态）
□ 能否自主决策（不依赖硬编码规则）
□ 能否执行行动（调用工具/输出结果）
□ 是否有目标导向（优化某个指标）

【非智能体示例】
- 纯规则引擎（if-else链）
- 固定流程的自动化脚本
- 单次调用的LLM API

【智能体示例】
- 能根据用户反馈调整策略的对话系统
- 能自主规划步骤并调用工具的任务执行器
- 多轮迭代优化的代码生成器

| 层级     | 判定条件          | 适用场景         | 技术选型建议                   |
| ------ | ------------- | ------------ | ------------------------ |

def execute_chain(input_data, step_config):
    context = {"input": input_data}
    for step_name, config in step_config.items():
        # 格式化提示词
        prompt = config["prompt_template"].format(**context)
        
        # 调用LLM
        response = llm_call(prompt)
        
        # 解析输出（必须验证格式）
        parsed = parse_and_validate(response, config["output_format"])
        if not parsed:
            raise ChainBreakError(f"步骤 {step_name} 输出格式错误")
        
        # 更新上下文
        context[step_name] = parsed
    
    return context["step3_synthesize"]

def route_input(input_data, config):
    # 分类
    route_key = classify(input_data, config["classifier"])
    
    # 获取处理配置
    route = config["routes"].get(route_key, config["routes"]["escalate"])
    
    # 前置检查
    if "pre_check" in route and not route["pre_check"](input_data):
        return {"error": "前置检查失败", "routed_to": "escalate"}
    
    # 执行
    return route["handler"](input_data, timeout=route["timeout"])

def parallel_sectioning(long_document, section_handler):
    # 分段（保持语义完整）
    sections = split_with_overlap(long_document, 
                                  chunk_size=1000, 
                                  overlap=100)
    
    # 并行处理
    with ThreadPoolExecutor() as executor:
        futures = [
            executor.submit(section_handler, section)
            for section in sections
        ]
        results = [f.result() for f in futures]
    
    # 合并结果（去重+排序）
    return merge_results(results, deduplicate=True)

def parallel_voting(query, models, aggregator):
    # 多模型并行调用
    with ThreadPoolExecutor() as executor:
        futures = [
            executor.submit(model.generate, query)
            for model in models
        ]
        responses = [f.result() for f in futures]
    
    # 聚合策略
    if aggregator == "majority":
        return majority_vote(responses)
    elif aggregator == "confidence":
        return weighted_by_confidence(responses)
    elif aggregator == "llm_judge":
        return llm_select_best(responses)

def reflection_loop(task_input, config):
    current_output = None
    
    for i in range(config["termination"]["max_iterations"]):
        # 生成
        if i == 0:
            current_output = producer.generate(task_input, config["producer"])
        else:
            current_output = producer.revise(current_output, feedback, config["producer"])
        
        # 评估
        critique = critic.evaluate(current_output, config["critic"])
        
        # 终止判断
        if should_stop(critique, config["termination"]):
            return {
                "output": current_output,
                "iterations": i + 1,
                "final_score": critique["score"]
            }
        
        feedback = critique
    
    return {"output": current_output, "iterations": i + 1, "status": "max_iter_reached"}

def react_executor(user_input, tools, max_steps=10):
    context = {"user_input": user_input, "history": []}
    
    for step in range(max_steps):
        # 构建提示
        prompt = build_react_prompt(context, tools)
        
        # 获取模型输出
        response = llm.generate(prompt)
        
        # 解析
        parsed = parse_react_response(response)
        
        if parsed["type"] == "final":
            return parsed["answer"]
        
        elif parsed["type"] == "action":
            # 执行工具
            tool_result = execute_tool(parsed["tool"], parsed["parameters"])
            
            # 更新上下文
            context["history"].append({
                "thought": parsed["thought"],
                "action": parsed["action"],
                "observation": tool_result
            })
    
    return {"error": "超过最大步数", "history": context["history"]}

def plan_and_execute(goal, config):
    # 生成初始计划
    plan = generate_plan(goal, config["plan_generation"])
    
    executed = []
    current_step = 0
    
    while current_step < len(plan):
        step = plan[current_step]
        
        # 检查依赖
        if not all(d in executed for d in step["dependencies"]):
            raise PlanError(f"步骤{step['step_id']}依赖未满足")
        
        # 执行
        try:
            result = execute_step(step)
            executed.append(step["step_id"])
            current_step += 1
            
            # 验证结果
            if not validate_result(result, step["validation"]):
                raise ValidationError(f"步骤{step['step_id']}验证失败")
            
        except Exception as e:
            # 触发重规划
            new_plan = replan(plan, executed, str(e), config["replanning"])
            plan = new_plan
            # 从失败步骤重新开始
            current_step = len(executed)
    
    return {"status": "success", "executed": executed}

def hierarchical_multi_agent(user_goal, config):
    coordinator = Agent(config["coordinator"])
    
    # 分解任务
    task_plan = coordinator.decompose(user_goal)
    
    # 创建workers
    workers = {
        name: Agent(worker_config) 
        for name, worker_config in config["workers"].items()
    }
    
    results = {}
    
    # 按依赖顺序执行
    for task in topological_sort(task_plan["subtasks"]):
        worker = workers[task["assignee"]]
        
        # 收集依赖输入
        inputs = {
            dep: results[dep] 
            for dep in task["dependencies"]
        }
        
        # 执行
        result = worker.execute(task, inputs)
        results[task["id"]] = result
        
        # 向coordinator汇报
        coordinator.report_progress(task["id"], result)
    
    # 整合最终结果
    final_output = coordinator.synthesize(results)
    return final_output

class MemoryManager:
    def __init__(self, config):
        self.session_service = config["session"]["storage"]()
        self.memory_service = config["long_term"]["storage"]()
    
    def get_context(self, session_id, query=None, max_tokens=4000):
        """获取当前上下文（三层融合）"""
        session = self.session_service.get_session(session_id)
        
        # 1. 获取Session历史
        recent_events = self._get_recent_events(session, max_tokens * 0.5)
        
        # 2. 获取State数据
        state_context = self._format_state(session.state)
        
        # 3. 检索长期记忆（如有查询）
        long_term_memories = []
        if query:
            long_term_memories = self.memory_service.search_memory(
                query=query,
                top_k=MEMORY_ARCHITECTURE["long_term"]["retrieval"]["top_k"]
            )
        
        # 4. 融合上下文（按优先级排序）
        context = self._merge_context(
            system_prompt=SYSTEM_PROMPT,
            long_term=long_term_memories,
            state=state_context,
            recent=recent_events
        )
        
        return context
    
    def update_memory(self, session_id, event, state_delta=None):
        """更新记忆（三层同步）"""
        # 更新Session
        session = self.session_service.get_session(session_id)
        session.events.append(event)
        
        # 更新State
        if state_delta:
            session.state.update(state_delta)
        
        # 异步更新长期记忆（重要事件）

        self.code = initial_code
        self.archive = [{"version": 0, "code": initial_code, "score": 0}]
        self.config = config
    
    def improve_loop(self):

            # 1. 选择最佳历史版本作为基础
            base_version = self._select_best_version()
            
            # 2. 分析改进机会
            improvement_opportunities = self._analyze_archive()
            
            # 3. 生成修改方案

                base_version, 
                improvement_opportunities
            )
            
            # 4. 应用修改
            new_code = self._apply_modification(base_version["code"], modification)
            
            # 5. 基准测试
            score = self._evaluate(new_code)

                self.code = new_code

class CircuitBreaker:
    def __init__(self, config):
        self.state = "CLOSED"  # CLOSED, OPEN, HALF_OPEN
        self.failure_count = 0
        self.success_count = 0
        self.last_failure_time = None
        self.config = config
    
    def call(self, func, *args, **kwargs):
        if self.state == "OPEN":
            if self._should_attempt_reset():
                self.state = "HALF_OPEN"
                self.success_count = 0
            else:
                raise CircuitBreakerOpen("服务暂时不可用，请稍后重试")
        
        try:
            result = func(*args, **kwargs)
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise
    
    def _on_success(self):
        if self.state == "HALF_OPEN":
            self.success_count += 1
            if self.success_count >= self.config["success_threshold"]:
                self.state = "CLOSED"
                self.failure_count = 0
        else:
            self.failure_count = 0
    
    def _on_failure(self):
        self.failure_count += 1
        self.last_failure_time = time.time()
        
        if self.state == "HALF_OPEN":
            self.state = "OPEN"
        elif self.failure_count >= self.config["failure_threshold"]:
            self.state = "OPEN"
    
    def _should_attempt_reset(self):
        return (time.time() - self.last_failure_time) > self.config["timeout_duration"]

def resilient_execute(max_retries=3, 
                      retry_exceptions=(TransientError,),
                      circuit_breaker=None,
                      fallback=None):
    """弹性执行装饰器"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            last_exception = None
            
            for attempt in range(max_retries):
                try:
                    # 尝试执行
                    if circuit_breaker:
                        return circuit_breaker.call(func, *args, **kwargs)
                    else:
                        return func(*args, **kwargs)
                        
                except retry_exceptions as e:
                    last_exception = e
                    # 指数退避
                    wait_time = 2 ** attempt
                    logger.warning(f"尝试{attempt+1}失败，{wait_time}s后重试: {e}")
                    time.sleep(wait_time)
                    
                except NonRetryableError as e:
                    # 不可重试错误，立即失败
                    logger.error(f"不可恢复错误: {e}")
                    raise
            
            # 重试耗尽，执行降级
            if fallback:
                logger.info("执行降级策略")
                return fallback(*args, **kwargs)
            
            raise MaxRetriesExceeded(f"重试{max_retries}次后仍失败") from last_exception
        
        return wrapper
    return decorator

                self._generate_query_variations(query)

            query, unique_results, top_k=5

class TreeOfThoughts:
    def __init__(self, config):
        self.config = config
    
    def solve(self, problem):
        """ToT求解"""
        root = ThoughtNode(content=problem, parent=None)
        current_level = [root]
        
        for depth in range(self.config["max_depth"]):
            next_level = []
            
            # 扩展每个节点
            for node in current_level:
                # 生成候选
                candidates = self._generate_candidates(node)
                
                for candidate in candidates:
                    child = ThoughtNode(
                        content=candidate,
                        parent=node,
                        depth=depth + 1
                    )
                    node.children.append(child)
                    next_level.append(child)
            
            # 评估并选择
            scored = [(n, self._evaluate(n)) for n in next_level]
            scored.sort(key=lambda x: x[1], reverse=True)
            
            # 剪枝：只保留top-k
            current_level = [n for n, _ in scored[:self.config["beam_width"]]]
            
            # 检查是否找到解
            for node in current_level:
                if self._is_solution(node):
                    return self._extract_path(node)
        
        # 返回最优路径
        best = max(current_level, key=lambda n: self._evaluate(n))
        return self._extract_path(best)
    
    def _generate_candidates(self, node):
        """生成候选思路"""
        prompt = f"""
        问题: {self.root.content}
        当前思路: {self._get_path(node)}
        
        请生成{self.config['branching_factor']}个不同的下一步思路。
        每个思路应该：
        - 基于当前进展
        - 探索不同方向
        - 具体可操作
        
        输出格式：每行一个思路
        """
        
        response = llm.generate(prompt)
        return [line.strip() for line in response.split("\n") if line.strip()]

class AgentEvaluator:

    def evaluate_trajectory(self, trajectory, expected=None):

        results = {}

        results["task_success"] = self._check_task_completion(
            trajectory, expected
        )
        
        # 步骤效率
        results["step_efficiency"] = len(trajectory["steps"]) / \
            self._estimate_optimal_steps(trajectory["task"])
        
        # 工具准确性
        tool_calls = [s for s in trajectory["steps"] if s["type"] == "tool_call"]
        correct_calls = sum(1 for t in tool_calls if t["status"] == "success")
        results["tool_accuracy"] = correct_calls / len(tool_calls) if tool_calls else 1.0
        
        # 延迟
        results["latency"] = trajectory["end_time"] - trajectory["start_time"]
        
        # 成本
        results["cost"] = sum(
            s.get("token_usage", 0) * s.get("model_price", 0)
            for s in trajectory["steps"]
        )
        
        return results

class PriorityQueue:
    def __init__(self, config):
        self.config = config
        self.queue = []
        self.in_progress = {}
    
    def calculate_priority(self, task):
        """计算任务优先级"""
        factors = self.config["factors"]
        
        urgency = self._calculate_urgency(task, factors["urgency"])
        importance = self._calculate_importance(task, factors["importance"])
        effort = self._calculate_effort(task, factors["effort"])
        dependencies = self._calculate_dependencies(task, factors["dependencies"])
        
        policy = self.config["queue_policies"].get(
            task.get("policy", "default")
        )
        
        priority = (
            urgency * factors["urgency"]["weight"] +
            importance * factors["importance"]["weight"] +
            effort * factors["effort"]["weight"] +
            dependencies * factors["dependencies"]["weight"]
        )
        
        # 饥饿预防提升
        if task.get("wait_time", 0) > self.config["scheduling"]["starvation_prevention"]["max_wait_time"]:
            boost = task["wait_time"] / self.config["scheduling"]["starvation_prevention"]["max_wait_time"]
            priority *= (1 + boost)
        
        return priority
    
    def enqueue(self, task):
        """加入队列"""
        task["priority"] = self.calculate_priority(task)
        task["enqueued_at"] = time.time()
        task["wait_time"] = 0
        
        heapq.heappush(self.queue, (-task["priority"], task["id"], task))
    
    def dequeue(self):
        """取出最高优先级任务"""
        if not self.queue:
            return None
        
        _, _, task = heapq.heappop(self.queue)
        
        # 检查抢占
        if self.in_progress and self.config["scheduling"]["preemption"]["enabled"]:
            current = list(self.in_progress.values())[0]
            if task["priority"] > current["priority"] * 1.5:
                self._preempt(current)
        
        self.in_progress[task["id"]] = task
        return task
